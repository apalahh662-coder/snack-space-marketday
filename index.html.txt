        function getAudioContext() {
            if (!audioCtx) {
                const AudioContextClass = window.AudioContext || window.webkitAudioContext;
                if (AudioContextClass) {
                    audioCtx = new AudioContextClass();
                }
            }
            if (audioCtx && audioCtx.state === 'suspended') {
                audioCtx.resume();
            }
            return audioCtx;
        }

        function unlockAudioOnMobile() {
            const ctx = getAudioContext();
            if (ctx && ctx.state === 'running') {
                window.removeEventListener('touchstart', unlockAudioOnMobile);
                window.removeEventListener('click', unlockAudioOnMobile);
            }
        }
        window.addEventListener('touchstart', unlockAudioOnMobile, { passive: true });

        window.playSound = function(type) {
            if (!soundEnabled) return;
            try {
                const ctx = getAudioContext();
                if (!ctx) return;

                const osc = ctx.createOscillator();
                const gain = ctx.createGain();
                osc.connect(gain);
                gain.connect(ctx.destination);
                const now = ctx.currentTime;

                if (type === 'coin') {
                    osc.type = 'square';
                    osc.frequency.setValueAtTime(987.77, now);
                    osc.frequency.setValueAtTime(1318.51, now + 0.08);
                    gain.gain.setValueAtTime(0.15, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.35);
                    osc.start(now);
                    osc.stop(now + 0.35);
                } else if (type === 'jump') {
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(150, now);
                    osc.frequency.exponentialRampToValueAtTime(600, now + 0.15);
                    gain.gain.setValueAtTime(0.2, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.15);
                    osc.start(now);
                    osc.stop(now + 0.15);
                } else if (type === 'success') {
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(523.25, now);
                    osc.frequency.setValueAtTime(659.25, now + 0.1);
                    osc.frequency.setValueAtTime(783.99, now + 0.2);
                    osc.frequency.setValueAtTime(1046.50, now + 0.3);
                    gain.gain.setValueAtTime(0.2, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.6);
                    osc.start(now);
                    osc.stop(now + 0.6);
                } else if (type === 'click') {
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(440, now);
                    gain.gain.setValueAtTime(0.05, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.05);
                    osc.start(now);
                    osc.stop(now + 0.05);
                }
            } catch (e) {
                console.log('Audio error:', e);
            }
        }

        async function initFirebaseApp() {
            try {
                if (typeof firebaseConfig !== 'undefined' && firebaseConfig.apiKey && firebaseConfig.apiKey !== 'mock') {
                    app = initializeApp(firebaseConfig);
                    db = getFirestore(app);
                    auth = getAuth(app);

        function setupFirestoreListeners() {
            if (!db) return;

            const productsRef = collection(db, 'artifacts', appId, 'public', 'data', 'products');
            onSnapshot(productsRef, (snapshot) => {
                const loadedProducts = [];
                snapshot.forEach(docSnap => {
                    loadedProducts.push({ id: docSnap.id, ...docSnap.data() });
                });
                if (loadedProducts.length > 0) {
                    products = loadedProducts;
                    localStorage.setItem('snack_space_products', JSON.stringify(products));
                } else {
                    products.forEach(p => {
                        setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'products', p.id), p);
                    });
                }
                renderBuyerProducts();
                if (typeof renderStockList === 'function') renderStockList();
            }, (error) => {
                console.error("Products subscription error:", error);
                enableLocalFallback();
            });

            const ordersRef = collection(db, 'artifacts', appId, 'public', 'data', 'orders');
            onSnapshot(ordersRef, (snapshot) => {
                const loadedOrders = [];
                snapshot.forEach(docSnap => {
                    loadedOrders.push({ id: docSnap.id, ...docSnap.data() });
                });
                loadedOrders.sort((a, b) => (b.orderNumber || 0) - (a.orderNumber || 0));
                orders = loadedOrders;
                localStorage.setItem('snack_space_orders', JSON.stringify(orders));
                if (isAdminLoggedIn) {
                    renderSellerDashboard();
                }
                updatePendingBadge();
            }, (error) => {
                console.error("Orders subscription error:", error);
            });
        }

        window.updateCart = function(productId, delta) {
            playSound('jump');
            const current = cart[productId] || 0;
            const product = products.find(p => p.id === productId);
            const newQty = current + delta;
            
            if (newQty < 0) return;
            if (product && product.maxStock !== undefined && newQty > product.maxStock) {
                showToast(`Stok ${product.name} tersisa ${product.maxStock} saja!`, 'error');
                return;
            }

            if (newQty === 0) {
                delete cart[productId];
            } else {
                cart[productId] = newQty;
            }
            renderBuyerProducts();
        }

        function updateStats() {
            let revenue = 0;
            let gross = 0;
            let itemsSoldCount = 0;
            let pendingCount = 0;

            orders.forEach(o => {
                gross += o.total;
                if (o.status === 'Selesai') {
                    revenue += o.total;
                    o.items.forEach(i => itemsSoldCount += i.qty);
                }
                if (o.status === 'Pending') {
                    pendingCount++;
                }
            });

            document.getElementById('stat-revenue').textContent = `Rp ${revenue.toLocaleString('id-ID')}`;
            document.getElementById('stat-gross-revenue').textContent = `Rp ${gross.toLocaleString('id-ID')}`;
            document.getElementById('stat-orders-count').textContent = orders.length;
            document.getElementById('stat-pending-count').textContent = pendingCount;
            document.getElementById('stat-items-sold').textContent = `${itemsSoldCount} Pcs`;

            const target = 500000;
            let percent = Math.min(100, Math.round((revenue / target) * 100));
            document.getElementById('target-percent').textContent = `${percent}%`;
            document.getElementById('target-progress-bar').style.width = `${percent}%`;
        }

        function renderProductSummary() {
            const grid = document.getElementById('product-summary-grid');
            grid.innerHTML = '';

            let productCounts = {};
            products.forEach(p => productCounts[p.name] = { count: 0, icon: p.icon });

            orders.forEach(o => {
                o.items.forEach(i => {
                    if (!productCounts[i.name]) productCounts[i.name] = { count: 0, icon: '📦' };
                    productCounts[i.name].count += i.qty;
                });
            });

            for (const [name, data] of Object.entries(productCounts)) {
                const itemBox = document.createElement('div');
                itemBox.className = 'bg-gray-50 border-2 border-black p-2.5 rounded-lg flex items-center justify-between';
                itemBox.innerHTML = `
                    <div class="flex items-center space-x-2">
                        <span class="text-xl">${data.icon}</span>
                        <span class="font-bold text-xs text-gray-800">${name}</span>
                    </div>
                    <span class="pixel-font text-xs bg-yellow-200 px-2 py-0.5 rounded border border-black">${data.count} Pcs</span>
                `;
                grid.appendChild(itemBox);
            }
        }

        function renderClassSummary() {
            const list = document.getElementById('class-summary-list');
            list.innerHTML = '';

            let classCounts = {};
            orders.forEach(o => {
                const cls = o.class || 'Lainnya';
                if (!classCounts[cls]) classCounts[cls] = 0;
                classCounts[cls] += o.total;
            });

            if (Object.keys(classCounts).length === 0) {
                list.innerHTML = `<div class="text-xs text-gray-500 text-center py-4">Belum ada data kelas.</div>`;
                return;
            }

            for (const [cls, total] of Object.entries(classCounts)) {
                const row = document.createElement('div');
                row.className = 'flex justify-between items-center bg-white p-2 rounded border border-yellow-300 text-xs font-bold';
                row.innerHTML = `<span><i class="fa-solid fa-graduation-cap text-yellow-600 mr-1.5"></i> ${cls}</span><span class="text-green-700">Rp ${total.toLocaleString('id-ID')}</span>`;
                list.appendChild(row);
            }
        }
